################################################################################
# Dockerfile for TASTY Visual Classification — GPU Retraining
################################################################################
#
# Optimized for:
#   - Ubuntu 24.04 LTS (Noble Numbat)
#   - NVIDIA RTX 6000 Ada Generation (48GB VRAM, Compute 8.9)
#   - NVIDIA Driver 580.x / CUDA 13.0 on host
#   - Docker 29.2.0 with NVIDIA Container Toolkit 1.18.2
#
# Build:
#   docker build -t tasty-training .
#
# Run (Jupyter server on port 8888):
#   docker run -it --rm --gpus all -p 8888:8888 \
#     -v "%cd%\checkpoint:/app/checkpoint" \
#     -v "%cd%\dataset:/app/dataset" \
#     tasty-training
#
# Then open in your browser:
#   http://localhost:8888  (token: tasty)
#
# Verify GPU inside the container:
#   docker run --rm --gpus all tasty-training python -c "import torch; print(torch.cuda.get_device_name(0))"
#
################################################################################

# Same verified base as your coworker's GPU Dockerfile.
# CUDA 12.8 runtime — PyTorch bundles its own cuDNN, so the plain runtime is enough.
FROM nvidia/cuda:12.8.0-runtime-ubuntu24.04

ENV DEBIAN_FRONTEND=noninteractive

WORKDIR /app

################################################################################
# GPU Passthrough
################################################################################
# "compute" enables CUDA/cuDNN for PyTorch training.
# "utility" enables nvidia-smi.
# (No need for "video" here — that's for NVENC/FFmpeg encoding.)
ENV NVIDIA_VISIBLE_DEVICES=all
ENV NVIDIA_DRIVER_CAPABILITIES=compute,utility

################################################################################
# STAGE 1: System Dependencies
################################################################################

RUN apt-get update && apt-get install -y --no-install-recommends \
    # Python runtime
    python3 \
    python3-pip \
    python3-venv \
    python3-dev \
    # Build tools (needed for some pip packages)
    build-essential \
    # OpenCV runtime libraries (headless — no display needed in a container)
    libgl1 \
    libglib2.0-0 \
    # Utilities
    curl \
    git \
    && rm -rf /var/lib/apt/lists/*

################################################################################
# STAGE 2: Python Virtual Environment
################################################################################
# We use a venv so pip installs are isolated and predictable.

ENV VIRTUAL_ENV=/opt/venv
RUN python3 -m venv $VIRTUAL_ENV
ENV PATH="$VIRTUAL_ENV/bin:$PATH"

################################################################################
# STAGE 3: PyTorch with CUDA
################################################################################
# We install the cu126 wheel (CUDA 12.6 build) — it is fully compatible with
# the CUDA 12.8 runtime in this container because CUDA is backward-compatible.
# DO NOT run: pip install torch  (that installs the CPU-only build by default)

RUN pip install --no-cache-dir \
    torch \
    torchvision \
    torchaudio \
    --index-url https://download.pytorch.org/whl/cu126

################################################################################
# STAGE 4: Training & Data Science Dependencies
################################################################################

RUN pip install --no-cache-dir \
    # Data handling
    numpy \
    pandas \
    # Image processing
    pillow \
    opencv-python-headless \
    # ML utilities
    scikit-learn \
    # Visualization
    matplotlib \
    seaborn \
    # Progress bars
    tqdm \
    # Jupyter environment (replaces Google Colab)
    jupyterlab \
    ipywidgets \
    ipykernel

################################################################################
# STAGE 5: Copy Project Files
################################################################################
# We only copy code — checkpoint models and the dataset are mounted as volumes
# at runtime so you don't bake large files into the image.

COPY scripts/ ./scripts/
COPY notebooks/ ./notebooks/

# Create the mount point directories so Docker knows they exist
RUN mkdir -p /app/checkpoint /app/dataset/labeled

################################################################################
# STAGE 6: Runtime Configuration
################################################################################

# Point training scripts to the correct dirs (override with -e flags if needed)
ENV CHECKPOINT_DIR=/app/checkpoint
ENV DATASET_DIR=/app/dataset
ENV PYTHONUNBUFFERED=1

# Expose JupyterLab port
EXPOSE 8888

################################################################################
# STAGE 7: Health Check & Entry Point
################################################################################

HEALTHCHECK --interval=30s --timeout=10s --start-period=30s --retries=3 \
    CMD curl -f http://localhost:8888/api || exit 1

# Start JupyterLab — same workflow as Colab, runs in your browser.
# Token is "tasty" — change it if the server is accessible to others.
CMD ["jupyter", "lab", \
     "--ip=0.0.0.0", \
     "--port=8888", \
     "--no-browser", \
     "--allow-root", \
     "--ServerApp.token=tasty", \
     "--notebook-dir=/app/notebooks"]

################################################################################
# How to use after launching:
#
#   1. Open http://localhost:8888 in your browser
#   2. Enter token: tasty
#   3. Open train_classifier_colab.ipynb
#   4. Update the dataset path at the top:
#        DATASET_DIR = "/app/dataset/labeled"
#        CHECKPOINT_DIR = "/app/checkpoint"
#   5. Run all cells — training will use the RTX 6000 GPU automatically
#
# Volume mounts explained:
#   -v "%cd%\checkpoint:/app/checkpoint"   <- reads existing model, writes new one here
#   -v "%cd%\dataset:/app/dataset"         <- your labeled images + CSV go here
#
# To export labels from the labeling tool before training:
#   Go to the labeling UI → Export → Wide CSV format → save to dataset/labeled/
#
# Verify GPU is detected (before starting the full training run):
#   import torch
#   print(torch.cuda.is_available())            # Should print: True
#   print(torch.cuda.get_device_name(0))        # Should print: NVIDIA RTX 6000 Ada
